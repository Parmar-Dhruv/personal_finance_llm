# --- Auth ---
# pwdlib, not passlib: passlib has been unmaintained since 2020, breaks on
# Python 3.13+, and broke again against bcrypt>=5.0 (Oct 2025). pwdlib is
# FastAPI's current official recommendation, defaulting to Argon2id.
pwdlib[argon2]
# PyJWT, not python-jose: python-jose's last release is ~3 years stale.
# PyJWT is actively maintained and is all we need (no JWKS/multi-IdP here).
pyjwt
