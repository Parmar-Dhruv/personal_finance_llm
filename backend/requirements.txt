# --- Document intelligence (Phase 2) ---
# pdfplumber, not PyMuPDF: PyMuPDF is AGPL-3.0 (or paid Artifex commercial
# license); pdfplumber is MIT. pdfplumber's char/bbox-level model is also
# the better fit for this project's doc->page->region provenance
# requirement, not just the safer license. Decided 2026-09-12.
pdfplumber
# pypdfium2, not Wand/ImageMagick (pdfplumber's own to_image()) or
# pdf2image (poppler): rasterizing scanned pages for OCR without pulling
# in Ghostscript, which is ALSO AGPL. pypdfium2 wraps Google's own PDFium
# (Apache-2.0/BSD-3-Clause) — zero copyleft anywhere in the pipeline.
pypdfium2
# pytesseract + system tesseract-ocr binary (installed via apt in
# Dockerfile): Apache 2.0, still the standard free OCR baseline in 2026.
# Confirmed current (5.x); known weakness on noisy/handwritten scans is
# an accepted gap for now, not a reason to reach for paid cloud OCR yet.
pytesseract
Pillow
pandas
openpyxl

# --- Auth ---
# pwdlib, not passlib: passlib has been unmaintained since 2020, breaks on
# Python 3.13+, and broke again against bcrypt>=5.0 (Oct 2025). pwdlib is
# FastAPI's current official recommendation, defaulting to Argon2id.
pwdlib[argon2]
# PyJWT, not python-jose: python-jose's last release is ~3 years stale.
# PyJWT is actively maintained and is all we need (no JWKS/multi-IdP here).
pyjwt

# --- Normalization (Phase 3) ---
# python-dateutil, not dateparser: already a transitive dep via pandas,
# permissively (dual Apache-2.0/BSD-3-Clause) licensed, and this project
# only needs structured dayfirst=True parsing, not dateparser's
# free-text/multi-language machinery (pulls in regex, pytz, tzlocal for
# capabilities not used here). Decided 2026-09-22, alongside the strict
# DD/MM/YYYY-only policy (see app/pipeline/normalize/dates.py).
python-dateutil
# rapidfuzz, not fuzzywuzzy/thefuzz: MIT vs GPL — same license-hygiene
# call already made for PyMuPDF/Ghostscript/Garage in earlier phases.
rapidfuzz

# --- ML layer (Phase 5) ---
# scikit-learn (BSD-3), numpy (BSD-3), joblib (BSD-3): classifier, Isolation
# Forest benchmark, model persistence. Versions confirmed current on PyPI
# 2026-10-26 (scikit-learn 1.9.1, numpy 2.5.x, joblib 1.6.0). A saved model is
# tied to the exact scikit-learn version that trained it (see classifier.py).
scikit-learn
numpy
joblib

google-genai
